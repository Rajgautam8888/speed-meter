<script>

/* CLOCK */
setInterval(()=>{
document.getElementById("clock").innerText=
new Date().toLocaleTimeString();
},1000);

/* VOICE AI */
function voice(){
let rec = new (window.SpeechRecognition||window.webkitSpeechRecognition)();
rec.lang="en-US";

rec.onresult=function(e){
let cmd=e.results[0][0].transcript.toLowerCase();

if(cmd.includes("speed")){
alert("Speed is 85 KM/H");
}
if(cmd.includes("navigation")){
window.open("https://maps.google.com");
}
}

rec.start();
}

/* SYSTEM */
let fuel=100;

/* GPS HUD */
function start(){

if(navigator.geolocation){

navigator.geolocation.watchPosition((pos)=>{

let speed=pos.coords.speed;
if(speed==null) speed=Math.random()*30;

let kmh=Math.round(speed*3.6);

/* SPEED */
document.getElementById("speed").innerText=kmh;

/* GEAR */
let gear="N";
if(kmh>10)gear="1";
if(kmh>30)gear="2";
if(kmh>40)gear="3";
if(kmh>45)gear="4";
if(kmh>55)gear="5";
document.getElementById("gear").innerText="GEAR "+gear;

/* RPM */
let rpm=kmh*150;
document.getElementById("rpm").innerText="RPM: "+rpm;

/* TEMP */
let temp=70+kmh/3;
document.getElementById("temp").innerText="TEMP: "+temp.toFixed(0)+"°C";

/* FUEL */
fuel-=0.02;
if(fuel<0)fuel=0;
document.getElementById("fuel").innerText="FUEL: "+fuel.toFixed(0)+"%";

/* COMPASS */
let dirs=["N","NE","E","SE","S","SW","W","NW"];
let d=dirs[Math.floor(Math.random()*8)];
document.getElementById("dir").innerText="DIR: "+d;

/* WARNING */
let warn="";
if(kmh>120) warn="⚠ OVERSPEED";
if(temp>95) warn="⚠ ENGINE HOT";
if(fuel<10) warn="⚠ LOW FUEL";

document.getElementById("warn").innerText=warn;

});

}

}

start();
